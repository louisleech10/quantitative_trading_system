```python
def _calculate_past_stability_features(
    self,
    df: pd.DataFrame,
    periods_24h: int,
    periods_48h: int,
    periods_72h: int
) -> pd.DataFrame:
    """
    計算分類特徵（改寫版本）
    
    ⚠️ 重要：從T-1 bar開始往前看，不包含T
    """
    try:
        # 計算3天的lookback_bars
        # 使用periods_72h作為3天的bar數
        lookback_bars = periods_72h
        
        # === 特徵1: 過去3天最大波動度 ===
        # 從T-1開始往前lookback_bars根
        # 計算 (High - Low) / Open
        volatilities = []
        for i in range(len(df)):
            if i < 1:  # T-1不存在
                volatilities.append(np.nan)
                continue
            
            # 從T-1往前取lookback_bars根（不包含T）
            start_idx = max(0, i - lookback_bars)
            end_idx = i  # 不包含i（當前T）
            
            if end_idx - start_idx < 1:
                volatilities.append(np.nan)
                continue
            
            window = df.iloc[start_idx:end_idx]
            bar_volatility = (window['high'] - window['low']) / window['open']
            max_vol = bar_volatility.max() * 100  # 轉為百分比
            volatilities.append(max_vol)
        
        df['past_24hr_max_single_move'] = volatilities
        
        # 分類
        df['volatility_class'] = df['past_24hr_max_single_move'].apply(
            lambda x: 'X' if pd.isna(x) else (
                'L' if x < 2 else (
                'M' if x < 5 else (
                'H' if x < 10 else 'X'
            )))
        )
        
        # === 特徵2: 過去3天方向性 ===
        # Close_T-1 到 Close_{T-1-lookback_bars} 的變化
        directions = []
        for i in range(len(df)):
            if i < 1:
                directions.append(np.nan)
                continue
            
            start_idx = max(0, i - lookback_bars)
            end_idx = i
            
            if end_idx - start_idx < 1:
                directions.append(np.nan)
                continue
            
            close_t1 = df.iloc[end_idx - 1]['close']  # T-1
            close_start = df.iloc[start_idx]['close']  # T-1往前lookback_bars
            
            if close_start > 0:
                direction = (close_t1 - close_start) / close_start * 100
                directions.append(direction)
            else:
                directions.append(np.nan)
        
        df['past_48hr_price_range'] = directions
        
        # 分類
        df['direction_class'] = df['past_48hr_price_range'].apply(
            lambda x: 'S' if pd.isna(x) else (
                'D' if x < -3 else (
                'S' if x < 3 else (
                'U' if x < 8 else 'V'
            )))
        )
        
        # === 特徵3: 過去3天量能穩定度 ===
        # CV = std / mean
        volume_cvs = []
        for i in range(len(df)):
            if i < 1:
                volume_cvs.append(np.nan)
                continue
            
            start_idx = max(0, i - lookback_bars)
            end_idx = i
            
            if end_idx - start_idx < 1:
                volume_cvs.append(np.nan)
                continue
            
            window = df.iloc[start_idx:end_idx]
            vol_mean = window['volume'].mean()
            vol_std = window['volume'].std()
            
            if vol_mean > 0:
                cv = vol_std / vol_mean
                volume_cvs.append(cv)
            else:
                volume_cvs.append(np.nan)
        
        df['past_72hr_avg_bar_volatility'] = volume_cvs
        
        # 分類
        df['volume_class'] = df['past_72hr_avg_bar_volatility'].apply(
            lambda x: 'B' if pd.isna(x) else (
                'A' if x < 0.5 else (
                'B' if x < 1.0 else 'C'
            ))
        )
        
        # === 清除不再使用的舊欄位 ===
        # 保留past_48hr_directional_movement和past_24hr_volume_stability為None
        # 以免API映射出錯
        df['past_48hr_directional_movement'] = np.nan
        df['past_24hr_volume_stability'] = np.nan
        
        self.logger.info("分類特徵計算完成（改寫版本）")
        
        return df
        
    except Exception as e:
        self.logger.error(f"計算分類特徵時出錯: {str(e)}", exc_info=True)
        # 確保欄位存在
        for col in ['past_24hr_max_single_move', 'past_48hr_price_range', 
                    'past_72hr_avg_bar_volatility', 'volatility_class',
                    'direction_class', 'volume_class']:
            if col not in df.columns:
                df[col] = np.nan
        return df
```

**說明**：
- ✓ 複用 `past_24hr_max_single_move` 存波動度數值
- ✓ 複用 `past_48hr_price_range` 存方向性數值  
- ✓ 複用 `past_72hr_avg_bar_volatility` 存量能CV
- ✓ 新增 `volatility_class`, `direction_class`, `volume_class` 存分類代碼
- ✓ 保留原函數名稱，調用處不需修改

```python
def _classify_market_state(self, volatility: str, direction: str, volume: str) -> tuple:
    """
    市場分類邏輯（按優先順序）
    
    Returns: (class_code, class_name, difficulty)
    """
    # C1: 平靜橫盤
    if volatility == 'L' and direction == 'S' and volume == 'A':
        return ('C1', '平靜橫盤', '中等')
    
    # C8: 縮量橫盤
    if volatility == 'L' and direction == 'S' and volume == 'C':
        return ('C8', '縮量橫盤', '中等')
    
    # C2: 正常橫盤
    if volatility in ['L', 'M'] and direction == 'S' and volume == 'B':
        return ('C2', '正常橫盤', '中等')
    
    # C6: 極端波動
    if volatility in ['H', 'X']:
        return ('C6', '極端波動', '簡單')
    
    # C9: 恐慌下跌
    if volatility in ['H', 'X'] and direction == 'D' and volume == 'C':
        return ('C9', '恐慌下跌', '簡單')
    
    # C3: 下跌趨勢
    if volatility in ['M', 'H'] and direction == 'D':
        return ('C3', '下跌趨勢', '簡單')
    
    # C5: 假突破
    if volatility == 'M' and direction == 'U' and volume == 'C':
        return ('C5', '假突破', '困難')
    
    # C7: 溫和上漲中
    if volatility == 'M' and direction == 'U' and volume in ['A', 'B']:
        return ('C7', '溫和上漲中', '困難')
    
    # C4: 高位震盪
    if volatility in ['M', 'H'] and direction in ['U', 'V'] and volume in ['B', 'C']:
        return ('C4', '高位震盪', '中等')
    
    # C10: 放量震盪
    if volatility in ['M', 'H'] and direction == 'S' and volume == 'C':
        return ('C10', '放量震盪', '中等')
    
    # C11/C12: 其他
    return ('C11', '其他組合', '混合')
```
